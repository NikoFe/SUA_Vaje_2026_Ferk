services:
  store_database:
    build:
      context: ./store_service/mysql
      dockerfile: Dockerfile
    container_name: store_mysql_db
    restart: always
    environment:
      MYSQL_ROOT_PASSWORD: ${DB_PASSWORD}
      MYSQL_DATABASE: ${MYSQL_STORE_DATABASE}
    ports:
      - "3301:3306"
    volumes:
      - store_db_data:/var/lib/mysql
    networks:
      - app_network
    healthcheck:
      test:
        [
          "CMD-SHELL",
          "mysqladmin ping -h 127.0.0.1 -uroot -p\"$${MYSQL_ROOT_PASSWORD}\" --silent",
        ]
      interval: 5s
      timeout: 5s
      retries: 12
      start_period: 20s
  store_app:
    build:
      context: ./store_service
      dockerfile: Dockerfile
    container_name: store_service
    ports:
      - "3001:3001"
    depends_on:
      store_database:
        condition: service_healthy
    environment:
      DB_HOST: store_database
      DB_USER: ${DB_USER}
      DB_PASSWORD: ${DB_PASSWORD}
      DB_NAME: ${DB_STORE_NAME}
    volumes:
      - ./store_service:/usr/src/app
